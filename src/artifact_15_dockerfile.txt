FROM node:20-alpine AS builder

WORKDIR /app
COPY worker/package*.json ./
RUN npm ci

COPY worker/ ./
RUN npm run build

FROM node:20-alpine AS runner
WORKDIR /app
ENV NODE_ENV=production

COPY worker/package*.json ./
RUN npm ci --only=production

COPY --from=builder /app/dist ./dist
USER node

CMD ["node", "dist/index.js"]
